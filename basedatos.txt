-- ÉPICA: USUARIOS Y SEGURIDAD (HU-01, HU-02, HU-03)

-- ============================================================

create table usuarios (

id uuid primary key references auth.users(id) on delete cascade,

nombre_completo varchar,

correo varchar unique not null,

avatar_url varchar,

esta_activo boolean default true,

creado_en timestamptz default now(),

actualizado_en timestamptz

);

comment on table usuarios is 'HU-01: Perfil público del usuario. La contraseña y verificación de correo las maneja auth.users de Supabase, no esta tabla.';

comment on column usuarios.id is 'Debe coincidir con el id de auth.users (Supabase Auth). Se crea vía trigger al registrarse, no manualmente.';

comment on column usuarios.nombre_completo is 'Nombre visible del usuario dentro de la plataforma (para mostrar en colaboración, comentarios, etc.)';

comment on column usuarios.correo is 'Copia de referencia del correo (el correo real y su verificación los gestiona auth.users). Útil para mostrarlo sin consultar el esquema auth.';

comment on column usuarios.avatar_url is 'URL de la foto de perfil, normalmente almacenada en Supabase Storage';

comment on column usuarios.esta_activo is 'Permite desactivar una cuenta sin borrarla (soft-disable), por ejemplo ante abuso o baja voluntaria';

comment on column usuarios.creado_en is 'Fecha de creación del perfil, igual a la fecha de registro';

comment on column usuarios.actualizado_en is 'Última vez que el usuario editó su perfil (nombre, avatar, etc.)';

alter table usuarios enable row level security;

create policy "usuarios_select_own_profile"

on usuarios for select

using (auth.uid() = id);

create policy "usuarios_insert_own_profile"

on usuarios for insert

with check (auth.uid() = id);

create policy "usuarios_update_own_profile"

on usuarios for update

using (auth.uid() = id)

with check (auth.uid() = id);

create table roles_sistema (

id uuid primary key default gen_random_uuid(),

usuario_id uuid not null references usuarios(id),

es_administrador boolean default false,

unique (usuario_id)

);

comment on table roles_sistema is 'HU-03: Rol de administrador a nivel de sistema (distinto del rol dentro de un proyecto, que vive en miembros_proyecto)';

comment on column roles_sistema.id is 'Identificador único de la asignación de rol de sistema';

comment on column roles_sistema.usuario_id is 'Usuario al que se le asigna el rol';

comment on column roles_sistema.es_administrador is 'true si el usuario es administrador general de toda la plataforma (no de un proyecto puntual)';

create table proyectos (

);

comment on table proyectos is 'HU-04: Crear, renombrar, archivar y eliminar proyectos';

comment on column proyectos.id is 'Identificador único del proyecto';

create table nodos (

id uuid primary key default gen_random_uuid(),

proyecto_id uuid not null references proyectos(id),

padre_id uuid references nodos(id),

nombre varchar not null,

tipo_nodo tipo_nodo not null,

tipo_archivo tipo_archivo,

contenido_actual text,

storage_path varchar,

tamano_bytes integer,

creado_por uuid references usuarios(id),

creado_en timestamptz default now(),

actualizado_en timestamptz

);

comment on table nodos is 'HU-05: Carpetas anidadas y archivos .tex/.bib/imágenes dentro de un proyecto. contenido_actual guarda el texto plano/LaTeX (cifrado); storage_path apunta a Supabase Storage para binarios.';

comment on column nodos.id is 'Identificador único del nodo (archivo o carpeta)';

comment on column nodos.proyecto_id is 'Proyecto al que pertenece este archivo o carpeta';

comment on column nodos.padre_id is 'Carpeta contenedora. NULL si está en la raíz del proyecto. Permite anidar carpetas indefinidamente.';

comment on column nodos.nombre is 'Nombre del archivo o carpeta, ej. "main.tex" o "imagenes"';

comment on column nodos.tipo_nodo is 'Indica si este registro es un archivo o una carpeta';

comment on column nodos.tipo_archivo is 'Si es un archivo, qué tipo es (.tex, .bib, imagen, otro). NULL si es carpeta.';

comment on column nodos.contenido_actual is 'Texto vigente del archivo (código LaTeX), guardado ya cifrado. Aplica solo a archivos de texto como .tex/.bib.';

comment on column nodos.storage_path is 'Ruta del archivo dentro de Supabase Storage, usada para binarios como imágenes o PDFs generados';

comment on column nodos.tamano_bytes is 'Tamaño del archivo en bytes, útil para mostrar en el explorador y controlar cuotas de almacenamiento';

comment on column nodos.creado_por is 'Usuario que creó este archivo o carpeta';

comment on column nodos.creado_en is 'Fecha de creación del nodo';

comment on column nodos.actualizado_en is 'Última vez que se modificó el contenido o los datos del nodo';
